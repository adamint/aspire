FROM rust:1.89-alpine AS build
WORKDIR /app
RUN apk add --no-cache musl-dev gcc
COPY . .
RUN --mount=type=cache,target=/usr/local/cargo/registry \
    cargo build --bin worker --release --target-dir /build/target && \
    bin="$(ls -1d /build/target/release/worker /build/target/*/release/worker 2>/dev/null | head -n1)" && \
    [ -n "$bin" ] || { echo "cargo build produced no binary under /build/target" >&2; exit 1; } && \
    mkdir -p /build/bin && \
    cp "$bin" /build/bin/worker

FROM alpine:3.22
RUN apk --no-cache add ca-certificates tzdata
RUN addgroup -S app && adduser -S -G app app
WORKDIR /app
COPY --from=build /build/bin/worker /app/worker
USER app
ENTRYPOINT ["/app/worker"]
