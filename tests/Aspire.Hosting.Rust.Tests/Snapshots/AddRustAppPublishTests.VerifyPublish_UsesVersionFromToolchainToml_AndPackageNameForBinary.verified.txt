FROM rust:1.89.0-alpine AS build
WORKDIR /app
RUN apk add --no-cache musl-dev gcc
COPY . .
RUN cargo build --release

FROM alpine:latest
RUN apk --no-cache add ca-certificates tzdata
RUN addgroup -S app && adduser -S -G app app
WORKDIR /app
COPY --from=build /app/target/release/aspire-sample-rust-app /app/aspire-sample-rust-app
USER app
ENTRYPOINT ["/app/aspire-sample-rust-app"]
