FROM rust:alpine AS build
WORKDIR /app
COPY . .
RUN --mount=type=cache,target=/usr/local/cargo/registry \
    cargo build --profile dist --target-dir /build/target && \
    bin="$(ls -1d /build/target/dist/my-service /build/target/*/dist/my-service 2>/dev/null | head -n1)" && \
    [ -n "$bin" ] || { echo "cargo build produced no binary under /build/target" >&2; exit 1; } && \
    mkdir -p /build/bin && \
    cp "$bin" /build/bin/my-service

FROM alpine:3.24
RUN (addgroup -S app || groupadd --system app) && (adduser -S -G app app || useradd --system --gid app --no-create-home app)
WORKDIR /app
COPY --from=build /build/bin/my-service /app/my-service
USER app
ENTRYPOINT ["/app/my-service"]
