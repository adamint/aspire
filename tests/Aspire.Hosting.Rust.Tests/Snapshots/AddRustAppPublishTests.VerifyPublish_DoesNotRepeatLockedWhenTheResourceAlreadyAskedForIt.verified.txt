FROM rust:1.89-alpine AS build
WORKDIR /app
COPY . .
RUN --mount=type=cache,target=/usr/local/cargo/registry \
    cargo build --locked --release --target-dir /build/target && \
    bin="$(ls -1d /build/target/release/my-service /build/target/*/release/my-service 2>/dev/null | head -n1)" && \
    [ -n "$bin" ] || { echo "cargo build produced no binary under /build/target" >&2; exit 1; } && \
    mkdir -p /build/bin && \
    cp "$bin" /build/bin/my-service

FROM alpine:3.22
RUN (addgroup -g 999 -S app || groupadd --system --gid 999 app) && (adduser -u 999 -S -G app app || useradd --system --uid 999 --gid 999 --no-create-home app)
WORKDIR /app
COPY --from=build /build/bin/my-service /app/my-service
USER app
ENTRYPOINT ["/app/my-service"]
