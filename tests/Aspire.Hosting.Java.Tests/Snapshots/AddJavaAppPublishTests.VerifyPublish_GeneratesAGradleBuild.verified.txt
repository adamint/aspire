FROM docker.io/library/gradle:8-jdk17 AS build
WORKDIR /app
ENV GRADLE_USER_HOME=/root/.gradle
COPY . .
RUN --mount=type=cache,id=aspire-java-gradle-api,target=/root/.gradle,sharing=locked \
    mkdir -p /build && gradle --no-daemon -x test build && jars=$(ls build/libs/*.jar 2>/dev/null | grep -Ev '(-plain|-sources|-javadoc)\.jar$' || true) && count=$(printf '%s\n' "$jars" | grep -c . || true) && if [ "$count" != "1" ]; then echo "Aspire: expected exactly one application JAR from the build of 'api' matching build/libs/*.jar, found $count:" >&2; echo "$jars" >&2; echo "Use WithJarArtifact(\"<relative path>\") to select one." >&2; exit 1; fi && cp $jars /build/app.jar

FROM docker.io/library/eclipse-temurin:17-jre
RUN groupadd --system --gid 999 app && useradd --system --gid 999 --uid 999 --no-create-home app
WORKDIR /app
COPY --from=build /build/app.jar /app/app.jar
USER app
ENTRYPOINT ["java","-jar","/app/app.jar"]
